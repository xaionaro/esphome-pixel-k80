cmake_minimum_required(VERSION 3.16)
project(yiscaxia_protocol_tests LANGUAGES C CXX)
enable_testing()
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
foreach(test controller_protocol controller_scheduler controller_policy available_pairs semantic_conversion tx_packet)
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_${test}.c")
    set(extension c)
  else()
    set(extension cpp)
  endif()
  add_executable(test_${test} tests/test_${test}.${extension})
  target_include_directories(test_${test} PRIVATE components/yiscaxia tests)
  target_compile_options(test_${test} PRIVATE -Wall -Wextra -Werror)
  target_link_libraries(test_${test} PRIVATE m)
  add_test(NAME ${test} COMMAND test_${test})
endforeach()
add_test(NAME channel_fairness COMMAND test_controller_policy fairness)

option(YISCAXIA_FULL_CHECKS "Test actual installed ESPHome SDK and schema/codegen" OFF)
if(YISCAXIA_FULL_CHECKS)
  find_program(ESPHOME_EXECUTABLE NAMES esphome)
  if(NOT ESPHOME_EXECUTABLE OR NOT EXISTS "${ESPHOME_EXECUTABLE}" OR IS_DIRECTORY "${ESPHOME_EXECUTABLE}")
    message(FATAL_ERROR "YISCAXIA_FULL_CHECKS requires ESPHOME_EXECUTABLE pointing to the ESPHome CLI")
  endif()
  execute_process(COMMAND "${ESPHOME_EXECUTABLE}" --version
      RESULT_VARIABLE cli_result OUTPUT_QUIET ERROR_QUIET TIMEOUT 10)
  if(NOT cli_result EQUAL 0)
    message(FATAL_ERROR "Configured ESPHome CLI failed --version")
  endif()
  include(tests/cmake/sdk_runtime.cmake)
  foreach(test IN ITEMS light_controller configuration_persistence md7105_transport)
    add_executable(test_${test} tests/test_${test}.cpp)
    if(NOT test STREQUAL "md7105_transport")
      target_sources(test_${test} PRIVATE components/yiscaxia/yiscaxia.cpp)
    endif()
    target_include_directories(test_${test} PRIVATE components/yiscaxia)
    target_compile_features(test_${test} PRIVATE cxx_std_20)
    target_compile_options(test_${test} PRIVATE -Wall -Wextra -Werror -Wno-missing-field-initializers)
    target_link_libraries(test_${test} PRIVATE yiscaxia_sdk_runtime m)
    add_test(NAME ${test} COMMAND test_${test})
  endforeach()
  target_sources(test_md7105_transport PRIVATE components/yiscaxia/md7105_transport.cpp)
  add_test(NAME schema_codegen COMMAND "${CMAKE_COMMAND}"
      -DPROJECT_DIR=${CMAKE_CURRENT_SOURCE_DIR}
      -DOUTPUT_DIR=${CMAKE_CURRENT_BINARY_DIR}/schema
      -DESPHOME_EXECUTABLE=${ESPHOME_EXECUTABLE}
      -P ${CMAKE_CURRENT_SOURCE_DIR}/tests/verify_schema.cmake)
else()
  message(STATUS "Pure protocol checks only; enable YISCAXIA_FULL_CHECKS for SDK/schema coverage")
endif()
